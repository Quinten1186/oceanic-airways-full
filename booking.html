function setupManagePage() { const form = document.getElementById("manage-form"); const resultBox = document.getElementById("manage-result"); if (!form || !resultBox) return; form.addEventListener("submit", (event) => { event.preventDefault(); const bookingRef = document.getElementById("bookingRefInput").value.trim(); const email = document.getElementById("bookingEmailInput").value.trim(); let bookings = []; try { bookings = JSON.parse(localStorage.getItem(BOOKINGS_KEY) || "[]"); } catch { bookings = []; } const booking = bookings.find( (entry) => entry.reference === bookingRef && String(entry.email).toLowerCase() === email.toLowerCase() ); if (!booking) { resultBox.innerHTML = `

No booking found. Please check your reference and email.

`; return; } resultBox.innerHTML = `

Booking details

Reference: ${booking.reference}

Passenger: ${booking.firstName} ${booking.lastName}

Flight: ${booking.flight.id} · ${airportMap[booking.flight.from]} → ${airportMap[booking.flight.to]}

Seat: ${booking.seat}

Total: ${money(booking.total)}

Status: ${booking.status}

`; }); }
