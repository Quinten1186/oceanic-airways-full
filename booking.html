const BOOKINGS_KEY = "oceanic-bookings"; function money(value) { return new Intl.NumberFormat("en-AU", { style: "currency", currency: "AUD", maximumFractionDigits: 0 }).format(value); } function createBookingCode() { return "OA-" + Math.floor(10000 + Math.random() * 90000); } function getSelectedFlight() { try { const raw = localStorage.getItem("oceanic-selected-flight"); return raw ? JSON.parse(raw) : null; } catch { return null; } } function calculateTotal(selectedFlight, formData) { let total = selectedFlight.price; if (formData.get("baggage")) total += 65; if (formData.get("meal")) total += 22; if (formData.get("insurance")) total += 38; return total; } function setupBookingPage() { const summaryBox = document.getElementById("booking-summary"); const totalBox = document.getElementById("booking-total"); const form = document.getElementById("booking-form"); if (!summaryBox || !form) return; const selectedFlight = getSelectedFlight(); if (!selectedFlight) { summaryBox.innerHTML = "

Please select a flight first.

"; form.style.display = "none"; return; } const airportMap = { BNE: "Brisbane", SYD: "Sydney", MEL: "Melbourne", PER: "Perth", LAX: "Los Angeles", SIN: "Singapore", DXB: "Dubai", HKG: "Hong Kong" }; summaryBox.innerHTML = `
${airportMap[selectedFlight.from]} ${selectedFlight.from}
${selectedFlight.duration}
${airportMap[selectedFlight.to]} ${selectedFlight.to}

Flight: ${selectedFlight.id} · ${selectedFlight.aircraft} · ${selectedFlight.time}

`; const updateTotal = () => { const formData = new FormData(form); totalBox.textContent = `Total: ${money(calculateTotal(selectedFlight, formData))}`; }; form.querySelectorAll("input, select").forEach((input) => { input.addEventListener("change", updateTotal); input.addEventListener("input", updateTotal); }); totalBox.textContent = `Total: ${money(selectedFlight.price)}`; form.addEventListener("submit", (event) => { event.preventDefault(); const formData = new FormData(form); const booking = { reference: createBookingCode(), flight: selectedFlight, firstName: formData.get("firstName"), lastName: formData.get("lastName"), email: formData.get("email"), phone: formData.get("phone"), type: formData.get("type"), seat: formData.get("seat"), baggage: !!formData.get("baggage"), meal: !!formData.get("meal"), insurance: !!formData.get("insurance"), total: calculateTotal(selectedFlight, formData), status: "Confirmed", createdAt: new Date().toISOString() }; const existing = JSON.parse(localStorage.getItem(BOOKINGS_KEY) || "[]"); existing.push(booking); localStorage.setItem(BOOKINGS_KEY, JSON.stringify(existing)); form.innerHTML = `

Booking confirmed

Your booking reference is ${booking.reference}.

We’ve sent your trip details to ${booking.email}.

Total paid: ${money(booking.total)}

Manage booking
`; }); } document.addEventListener("DOMContentLoaded", () => { setupBookingPage(); });
